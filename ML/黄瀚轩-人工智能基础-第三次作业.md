### 1. Python 实现代码

```python
import numpy as np
import matplotlib.pyplot as plt
import random


N_CITIES = 8        
POP_SIZE = 100     
CROSS_RATE = 0.8    
MUTATE_RATE = 0.02  
N_GENERATIONS = 200 # 迭代次数


random.seed(42)
np.random.seed(42)


# 随机生成8个城市的坐标
city_coords = np.random.rand(N_CITIES, 2) * 100 

# 计算城市间的距离矩阵
def get_dist_matrix(coords):
    dist_matrix = np.zeros((N_CITIES, N_CITIES))
    for i in range(N_CITIES):
        for j in range(N_CITIES):
            dist_matrix[i][j] = np.sqrt(np.sum((coords[i] - coords[j])**2))
    return dist_matrix

dist_matrix = get_dist_matrix(city_coords)


class GA:
    def __init__(self, n_cities):
        self.n_cities = n_cities
        self.pop = np.array([np.random.permutation(n_cities) for _ in range(POP_SIZE)])

    
    def get_fitness(self):
        fitness = np.zeros(POP_SIZE)
        for i, path in enumerate(self.pop):
            dist = 0
            for j in range(self.n_cities - 1):
                dist += dist_matrix[path[j], path[j+1]]
            dist += dist_matrix[path[-1], path[0]] # 回到起点
            fitness[i] = 1 / dist # 路径越短，适应度越高
        return fitness

    # 选择
    def select(self, fitness):
        idx = np.random.choice(np.arange(POP_SIZE), size=POP_SIZE, replace=True, p=fitness/fitness.sum())
        return self.pop[idx]

    # 交叉
    def crossover(self, parent, pop):
        if np.random.rand() < CROSS_RATE:
            i_ = np.random.randint(0, POP_SIZE, size=1)
            mother = pop[i_].flatten()
            cp1, cp2 = sorted(random.sample(range(self.n_cities), 2)) # 随机选择两个交叉点
            
            child = [-1] * self.n_cities
            child[cp1:cp2] = parent[cp1:cp2]
            
            
            p2_remain = [item for item in mother if item not in child]
            idx = 0
            for i in range(self.n_cities):
                if child[i] == -1:
                    child[i] = p2_remain[idx]
                    idx += 1
            return np.array(child)
        return parent

    # 变异
    def mutate(self, child):
        for i in range(self.n_cities):
            if np.random.rand() < MUTATE_RATE:
                swap_idx = np.random.randint(0, self.n_cities)
                child[i], child[swap_idx] = child[swap_idx], child[i]
        return child

    # 进化
    def evolve(self):
        fitness = self.get_fitness()
        best_idx = np.argmax(fitness)
        best_path = self.pop[best_idx]
        best_dist = 1 / fitness[best_idx]
        
        selected_pop = self.select(fitness)
        next_pop = []
        for parent in selected_pop:
            child = self.crossover(parent, selected_pop)
            child = self.mutate(child)
            next_pop.append(child)
        self.pop = np.array(next_pop)
        return best_path, best_dist


ga = GA(N_CITIES)
best_dists = []

plt.ion() 
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))

for gen in range(N_GENERATIONS):
    best_path, best_dist = ga.evolve()
    best_dists.append(best_dist)
    
    if gen % 20 == 0 or gen == N_GENERATIONS - 1:
        ax1.cla()
        ax2.cla()
        
        # 绘制路径图
        coords = city_coords[best_path]
        coords = np.vstack([coords, coords[0]]) # 闭合路径
        ax1.plot(coords[:, 0], coords[:, 1], 'co-')
        for i, txt in enumerate(best_path):
            ax1.annotate(f"City {txt}", (city_coords[txt][0], city_coords[txt][1]))
        ax1.set_title(f"Gen: {gen} | Best Distance: {best_dist:.2f}")
        
        # 绘制收敛曲线
        ax2.plot(best_dists, 'r-')
        ax2.set_xlabel("Generation")
        ax2.set_ylabel("Shortest Distance")
        ax2.set_title("Convergence Curve")
        
        plt.pause(0.1)

plt.ioff()
print(f"最终最短路径顺序: {best_path}")
print(f"最终最短路径长度: {best_dist:.2f}")
plt.show()
```
