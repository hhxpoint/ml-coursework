### 课后习题 9.4 K-均值聚类：不同k值与初始中心的DB指标、Dunn指标计算

**题目要求**: 试编程实现k均值算法，设置三组不同的k值、三组不同初始中心点，在西瓜数据集4.0上进行实验比较，并讨论什么样的初始中心有利于取得好结果。

### 1. Python 实现代码

```python
import numpy as np
import matplotlib.pyplot as plt
from sklearn.metrics import pairwise_distances
from itertools import combinations

# 西瓜数据集 4.0 
X = np.array([
    [0.697, 0.460], [0.774, 0.376], [0.634, 0.264], [0.608, 0.318], [0.556, 0.215],
    [0.403, 0.237], [0.481, 0.149], [0.437, 0.211], [0.666, 0.091], [0.243, 0.267],
    [0.245, 0.057], [0.343, 0.099], [0.639, 0.161], [0.657, 0.198], [0.360, 0.370],
    [0.593, 0.042], [0.719, 0.103], [0.359, 0.188], [0.339, 0.241], [0.282, 0.257],
    [0.748, 0.232], [0.714, 0.346], [0.483, 0.312], [0.478, 0.437], [0.525, 0.369],
    [0.751, 0.489], [0.532, 0.472], [0.473, 0.376], [0.725, 0.445], [0.446, 0.459]
])


class KMeans:
    def __init__(self, k, init_centers=None, max_iter=100, tol=1e-4):
        self.k = k
        self.init_centers = init_centers
        self.max_iter = max_iter
        self.tol = tol
        self.centers = None
        self.labels = None
        self.inertia_ = None
      
    def fit(self, X):
        n_samples = X.shape[0]
      
        # 初始化聚类中心
        if self.init_centers is not None:
            self.centers = np.array(self.init_centers)
        else:
            idx = np.random.choice(n_samples, self.k, replace=False)
            self.centers = X[idx].copy()
      
        for _ in range(self.max_iter):
            # 分配样本到最近的中心
            distances = pairwise_distances(X, self.centers, metric='euclidean')
            self.labels = np.argmin(distances, axis=1)
          
            # 更新聚类中心
            new_centers = np.zeros_like(self.centers)
            for i in range(self.k):
                cluster_points = X[self.labels == i]
                if len(cluster_points) > 0:
                    new_centers[i] = cluster_points.mean(axis=0)
                else:
                    new_centers[i] = self.centers[i]
          
            # 检查收敛
            center_shift = np.linalg.norm(new_centers - self.centers)
            self.centers = new_centers
          
            if center_shift < self.tol:
                break
      
        # 计算SSE
        self.inertia_ = sum(
            np.sum((X[self.labels == i] - self.centers[i]) ** 2)
            for i in range(self.k) if np.sum(self.labels == i) > 0
        )
      
        return self


def davies_bouldin_index(X, labels, centers):
    n_clusters = len(centers)
    if n_clusters < 2:
        return 0
  
  
    avg_distances = np.zeros(n_clusters)
    for i in range(n_clusters):
        cluster_points = X[labels == i]
        if len(cluster_points) > 1:
            avg_distances[i] = np.mean(np.linalg.norm(cluster_points - centers[i], axis=1))
  
    # 计算DB指标
    db_sum = 0
    for i in range(n_clusters):
        max_ratio = 0
        for j in range(n_clusters):
            if i != j:
                inter_dist = np.linalg.norm(centers[i] - centers[j])
                if inter_dist > 0:
                    ratio = (avg_distances[i] + avg_distances[j]) / inter_dist
                    max_ratio = max(max_ratio, ratio)
        db_sum += max_ratio
  
    return db_sum / n_clusters


def dunn_index(X, labels, centers):
    n_clusters = len(centers)
    if n_clusters < 2:
        return 0
  
  
    max_intra_dists = []
    for i in range(n_clusters):
        cluster_points = X[labels == i]
        if len(cluster_points) > 1:
            intra_dists = pairwise_distances(cluster_points, metric='euclidean')
            max_intra_dists.append(np.max(intra_dists))
        else:
            max_intra_dists.append(0)
  
    max_intra_dist = max(max_intra_dists) if max_intra_dists else 1e-10
  
  
    min_inter_dist = float('inf')
    for i, j in combinations(range(n_clusters), 2):
        points_i = X[labels == i]
        points_j = X[labels == j]
        if len(points_i) > 0 and len(points_j) > 0:
            inter_dists = pairwise_distances(points_i, points_j, metric='euclidean')
            min_inter_dist = min(min_inter_dist, np.min(inter_dists))
  
    if max_intra_dist == 0 or min_inter_dist == float('inf'):
        return 0
  
    return min_inter_dist / max_intra_dist


def run_experiments():
    k_values = [2, 3, 4]
  
    init_strategies = [
        {"name": "随机选择", "centers": None},  # 随机初始化
        {"name": "边缘点选择", "centers": {  # 选择相距较远的点
            2: [[0.243, 0.267], [0.774, 0.376]],
            3: [[0.243, 0.267], [0.774, 0.376], [0.656, 0.091]],
            4: [[0.243, 0.267], [0.774, 0.376], [0.656, 0.091], [0.360, 0.370]]
        }},
        {"name": "密集区选择", "centers": {  # 选择数据密集区域
            2: [[0.556, 0.215], [0.639, 0.161]],
            3: [[0.556, 0.215], [0.639, 0.161], [0.608, 0.318]],
            4: [[0.556, 0.215], [0.639, 0.161], [0.608, 0.318], [0.697, 0.460]]
        }}
    ]
  
    results = []
  
    for k in k_values:
        for strategy in init_strategies:
            if strategy["centers"] is None:
                km = KMeans(k=k, init_centers=None)
            else:
                km = KMeans(k=k, init_centers=strategy["centers"][k])
          
            km.fit(X)
          
            db = davies_bouldin_index(X, km.labels, km.centers)
            dunn = dunn_index(X, km.labels, km.centers)
          
            results.append({
                'k': k,
                'strategy': strategy["name"],
                'DB': db,
                'Dunn': dunn,
                'SSE': km.inertia_,
                'labels': km.labels.copy(),
                'centers': km.centers.copy()
            })
  
    return results



results = run_experiments()

for k in [2, 3, 4]:
    print(f"\n【k = {k}】")
    for r in results:
        if r['k'] == k:
            print(f"  {r['strategy']}: DB={r['DB']:.4f}, Dunn={r['Dunn']:.4f}, SSE={r['SSE']:.4f}")
```

### 3. 分析与结论

#### 3.1 指标解释

**DB指标 (Davies-Bouldin Index)**:
 计算公式: $DB = \frac{1}{k} \sum_{i=1}^{k} \max_{j != i} \left( \frac{s_i + s_j}{d_{ij}} \right)$
 $s_i$: 第i个簇内样本到中心的平均距离（簇内紧密度）
 $d_{ij}$: 簇i和簇j中心之间的距离（簇间分离度）
 **指标含义**: DB越小越好，表示簇内紧凑且簇间分离明显

**Dunn指标 (Dunn Index)**:
 计算公式: $Dunn = \frac{\min_{i != j} \{ d(i,j) \}}{\max_{k} \{ diam(k) \}}$
 分子: 任意两簇之间的最小距离
 分母: 最大簇的直径（簇内最远距离）
 **指标含义**: Dunn越大越好，表示簇间距离大且簇内紧致

#### 3.2 实验结论

1. **k值选择**:
   **k=2**: 将西瓜分为好瓜/坏瓜两类，适合二分类问题，指标表现稳定
   **k=3**: 能发现数据中的中间状态样本，聚类效果较优
   **k=4**: 可能过度划分，某些簇样本过少，指标相对较差
2. **初始中心策略影响**:
   **随机选择**: 结果不稳定，容易陷入局部最优，SSE波动较大
   **边缘点选择**: 选择相距较远的数据点作为初始中心，有利于发现明显的类别分界，通常能取得更好的DB和Dunn指标
   **密集区选择**: 在数据密集区域选择初始中心，收敛速度快，但可能遗漏边界样本
3. **什么样的初始中心有利于取得好结果**:
   **分散性**: 初始中心应分散在数据空间的不同区域，避免过于集中
   **代表性**: 尽量选择有代表性的数据点（如边缘点或局部密度中心）
   **距离最大化**: 初始中心之间的相互距离应尽可能大
   对于西瓜数据集，**"边缘点选择"**策略配合**k=3**通常能取得最佳的聚类效果
